cmake_minimum_required(VERSION 3.24)
project(LinkStandardCache LANGUAGES C CXX)

if(NOT MODULE_FILE OR NOT PROBE_FUNCTION)
  message(FATAL_ERROR "A link helper module and function are required")
endif()

# A preloaded check module must not choose the helper's actual policy scope.
if(PRELOAD_OLD)
  cmake_policy(PUSH)
  cmake_policy(SET CMP0067 OLD)
  include(CheckCXXSourceCompiles)
  cmake_policy(POP)
else()
  include(CheckCXXSourceCompiles)
endif()
include("${MODULE_FILE}")
if(CALLER_OLD)
  cmake_policy(SET CMP0067 OLD)
endif()
set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)
set(CMAKE_REQUIRED_DEFINITIONS -DSTANDARD_CACHE_STATE=1)
set(prefix METIS)
if(PROBE_FUNCTION STREQUAL "gklib_check_link")
  set(prefix GKLIB)
endif()

function(probe source language expected label cache_mode)
  get_cmake_property(before CACHE_VARIABLES)
  list(FILTER before INCLUDE REGEX "^${prefix}_LINK_CHECK_")
  cmake_language(CALL ${PROBE_FUNCTION} "${source}" actual LANGUAGE ${language})
  get_cmake_property(after CACHE_VARIABLES)
  list(FILTER after INCLUDE REGEX "^${prefix}_LINK_CHECK_")
  list(LENGTH before before_count)
  list(LENGTH after after_count)
  math(EXPR added "${after_count} - ${before_count}")
  if(NOT "${CMAKE_TRY_COMPILE_TARGET_TYPE}" STREQUAL "STATIC_LIBRARY" OR
      NOT "${CMAKE_REQUIRED_DEFINITIONS}" STREQUAL "-DSTANDARD_CACHE_STATE=1")
    message(FATAL_ERROR "The link helper leaked its check state")
  endif()
  if((expected AND NOT actual) OR (NOT expected AND actual))
    message(FATAL_ERROR "STANDARD_CACHE_MISMATCH: ${language} ${label}, expected=${expected}, actual=${actual}")
  endif()
  if((cache_mode STREQUAL "fresh" AND NOT added EQUAL 1) OR
      (cache_mode STREQUAL "cached" AND NOT added EQUAL 0))
    message(FATAL_ERROR "STANDARD_CACHE_KEY: ${language} ${label}, new keys=${added}")
  endif()
  file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/observations.txt"
    "${language}|${label}|expected=${expected}|actual=${actual}|new-keys=${added}|standard=${CMAKE_${language}_STANDARD}|required=${CMAKE_${language}_STANDARD_REQUIRED}|extensions=${CMAKE_${language}_EXTENSIONS}\n")
  message(STATUS "STANDARD_CACHE_RESULT ${language} ${label}: ${actual}; new keys=${added}")
endfunction()

foreach(language IN ITEMS C CXX)
  if(language STREQUAL "CXX")
    set(high 20)
    set(low 11)
    set(source "#ifdef _MSVC_LANG\n#define STD_VERSION _MSVC_LANG\n#else\n#define STD_VERSION __cplusplus\n#endif\n#if STD_VERSION < 202002L\n#error REQUIRES_CXX20\n#endif\nconsteval int value() { return 0; }\nint main() { return value(); }\n")
  else()
    set(high 17)
    set(low 11)
    set(source "#if !defined(__STDC_VERSION__) || __STDC_VERSION__ < 201710L\n#error REQUIRES_C17\n#endif\nint main(void) { return 0; }\n")
  endif()
  set(CMAKE_${language}_STANDARD_REQUIRED ON)
  set(CMAKE_${language}_EXTENSIONS OFF)
  if(PROBE_MODE STREQUAL "cycle")
    set(CMAKE_${language}_STANDARD ${high})
    probe("${source}" ${language} TRUE high any)
    set(CMAKE_${language}_STANDARD ${low})
    probe("${source}" ${language} FALSE low fresh)
    set(CMAKE_${language}_STANDARD ${high})
    probe("${source}" ${language} TRUE restored-high cached)

    # Supported standards need no fallback here. Required mode has structural
    # cache evidence; no semantic failure is invented when both modes work.
    set(plain "int main(void) { return 0; }\n")
    unset(CMAKE_${language}_STANDARD_REQUIRED)
    probe("${plain}" ${language} TRUE required-unset fresh)
    set(CMAKE_${language}_STANDARD_REQUIRED OFF)
    probe("${plain}" ${language} TRUE required-off fresh)
    set(CMAKE_${language}_STANDARD_REQUIRED ON)
    probe("${plain}" ${language} TRUE required-on fresh)

    set(extension_source "int main(void) { return 1 == 1 ? 0 : 1; }\n")
    set(strict_mode FALSE)
    if(NOT MSVC AND CMAKE_${language}_COMPILER_ID MATCHES "GNU|Clang")
      set(strict_mode TRUE)
      set(extension_source "#ifndef __STRICT_ANSI__\n#error STRICT_MODE_REQUIRED\n#endif\n${extension_source}")
    endif()
    unset(CMAKE_${language}_EXTENSIONS)
    set(expected TRUE)
    if(strict_mode AND CMAKE_${language}_EXTENSIONS_DEFAULT)
      set(expected FALSE)
    endif()
    probe("${extension_source}" ${language} ${expected} extensions-unset fresh)
    set(CMAKE_${language}_EXTENSIONS OFF)
    probe("${extension_source}" ${language} TRUE extensions-off fresh)
    set(CMAKE_${language}_EXTENSIONS ON)
    set(expected TRUE)
    if(strict_mode)
      set(expected FALSE)
    endif()
    probe("${extension_source}" ${language} ${expected} extensions-on fresh)
    message(STATUS "STANDARD_CACHE_EXTENSION_SEMANTICS ${language}: GNU-strict-control=${strict_mode}")

    set(default_source "int main(void) { return 2 == 2 ? 0 : 1; }\n")
    unset(CMAKE_${language}_STANDARD)
    unset(CMAKE_${language}_STANDARD_REQUIRED)
    unset(CMAKE_${language}_EXTENSIONS)
    probe("${default_source}" ${language} TRUE defaults-unset fresh)
    set(CMAKE_${language}_STANDARD "")
    probe("${default_source}" ${language} TRUE standard-empty fresh)
    unset(CMAKE_${language}_STANDARD)
    probe("${default_source}" ${language} TRUE defaults-restored cached)
  else()
    set(CMAKE_${language}_STANDARD ${high})
    set(expected TRUE)
    if(PROBE_MODE STREQUAL "low")
      set(CMAKE_${language}_STANDARD ${low})
      set(expected FALSE)
    endif()
    probe("${source}" ${language} ${expected} "reconfigure-${PROBE_MODE}" any)
  endif()
endforeach()
